#include <iostream>

#include <string>
#include <curl/curl.h>

// Helper callback for libcurl
static size_t WriteCallback(void* contents, size_t size, size_t nmemb, void* userp) {
    size_t total = size * nmemb;
    static_cast<std::string*>(userp)->append(static_cast<char*>(contents), total);
    return total;
}

// The simple function: pass your prompt, get back the response string
std::string askChatbot(const std::string& prompt) {
    CURL* curl = curl_easy_init();
    if (!curl) return "Error: Could not initialize cURL";

    std::string response;
    
    // Hardcoded JSON payload (escaping quotes inside prompt if needed, 
    // but for simple strings this works fine)
    std::string payload = 
        "{\"model\":\"~openai/gpt-sol-latest\","
        "\"messages\":[{\"role\":\"user\",\"content\":\"" + prompt + "\"}]}";

    struct curl_slist* headers = NULL;
    headers = curl_slist_append(headers, "Content-Type: application/json");
    headers = curl_slist_append(headers, "Authorization: Bearer YOUR_API_KEY");

    curl_easy_setopt(curl, CURLOPT_URL, "https://openrouter.ai/api/v1/chat/completions");
    curl_easy_setopt(curl, CURLOPT_POSTFIELDS, payload.c_str());
    curl_easy_setopt(curl, CURLOPT_HTTPHEADER, headers);
    curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, WriteCallback);
    curl_easy_setopt(curl, CURLOPT_WRITEDATA, &response);

    CURLcode res = curl_easy_perform(curl);
    if (res != CURLE_OK) {
        response = "Error: " + std::string(curl_easy_strerror(res));
    }

    curl_easy_cleanup(curl);
    curl_slist_free_all(headers);
    
    return response;
}

int main() {
    std::string reply = askChatbot("What is the meaning of life?");
    std::cout << reply << std::endl;
    return 0;
}
